¡¡ PRUEBA DE RECUPERACION DE ERRORES

empty ~ __main__ ~ <| |>
|:
    ¡¡ ERROR LEXICO: caracter no reconocido
    int ~ x <- 10 ! $ % &
    
    ¡¡ ERROR SINTACTICO: falta separador '~'
    int y <- 20 !
    
    ¡¡ ERROR SINTACTICO: falta delimitador '!'
    int ~ z <- 30
    
    ¡¡ ERROR SEMANTICO: variable no declarada
    w <- x + y !
    
    ¡¡ ERROR SEMANTICO: tipo incompatible
    float ~ precio <- 100 !
    
    ¡¡ CODIGO CORRECTO (debe ejecutarse despues de los errores)
    int ~ a <- 5 !
    int ~ b <- 10 !
    int ~ c <- a + b !
    cout ~ c !
:|
