¡¡ Funciones que se llaman entre si
int ~ esParHelper <| int ~ n |> |:
    return ~ n % 2 !
:|

bool ~ esPar <| int ~ n |> |:
    return ~ equal <| esParHelper <| n |>, 0 |> !
:|

bool ~ esImpar <| int ~ n |> |:
    return ~ equal <| esPar <| n |>, false |> !
:|
